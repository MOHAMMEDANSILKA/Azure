dept_id INT PRIMARY KEY,

);


CREATE TABLE employees (
  emp_id      INT PRIMARY KEY,
  first_name  VARCHAR(50) NOT NULL,
  last_name   VARCHAR(50) NOT NULL,
  email       VARCHAR(150) UNIQUE,
  salary      DECIMAL(12,2),
  hire_date   DATE,
  dept_id     INT,
  CONSTRAINT fk_emp_dept FORIGN KEY (dept_id) REFERENCES     departments(dept_id),CONSTRAINT chk_salary CHECK (salary >= 0)
);
